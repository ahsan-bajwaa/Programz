#include <iostream>
using namespace std;

bool isPrime(int number){
    if(number <= 1){
        return false;
    }
    for(int i = 2; i * i <= number; i++){
        if(number % i == 0){
            return false;
        }
    }
    return true;
}

int main(){
    int num;

    cout << "Enter number: ";
    cin >> num;

    if(isPrime(num)){
        cout << num << " is prime." << endl;
    }
	else{
        cout << num << " is not prime." << endl;
    }
    return 0;
}

#include <iostream>
using namespace std;

bool isPalindrome(int number){
    int original = number, reversed = 0, remainder;
    
	while(number > 0){
        remainder = number % 10;
        reversed = reversed * 10 + remainder;
        number /= 10;
    }
    return original == reversed;
}

int main(){
    int num;
    cout << "Enter number: ";
    cin >> num;

    if(isPalindrome(num)){
        cout << num << " is palindrome.";
    }
	else{
        cout << num << " is not palindrome.";
    }
    return 0;
}

#include <iostream>
using namespace std;

void calculatePower(int base, int exponent){
    int result = 1;
    
    for(int i = 1; i <= exponent; i++){
        result *= base;
    }
    cout << base << " raised of power " << exponent << " is = " << result << endl;
}

int main(){
    int base, exponent;
    cout << "Enter base: ";
    cin >> base;
    cout << "Enter exponent: ";
    cin >> exponent;
    
    calculatePower(base, exponent);
    return 0;
}

void printFactors(int num) {
    cout << "Factors of " << num << " are: ";
    
    for(int i = 1; i <= num; i++) {
        if(num % i == 0){
            cout << i << " ";
        }
    }
    cout << endl;
}

int main() {
    int num;
    cout << "Enter number: ";
    cin >> num;
    
    printFactors(num);
    return 0;
